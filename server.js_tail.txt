const server = http.createServer((req, res) => {
    // Enable CORS
    res.setHeader('Access-Control-Allow-Origin', '*');
    res.setHeader('Access-Control-Allow-Methods', 'GET, POST, DELETE, OPTIONS');
    res.setHeader('Access-Control-Allow-Headers', 'Content-Type');

    if (req.method === 'OPTIONS') {
        res.writeHead(204);
        res.end();
        return;
    }

    const url = req.url;

    // --- API ENDPOINTS ---
    if (url === '/api/account' && req.method === 'GET') {
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(db.account));
        return;
    }

    if (url === '/api/account/update' && req.method === 'POST') {
        let body = '';
        req.on('data', chunk => body += chunk);
        req.on('end', () => {
            const data = JSON.parse(body);
            if (data.customer_name) db.account.customer_name = data.customer_name;
            if (data.iban) db.account.iban = data.iban;
            if (data.balance !== undefined) db.account.balance = data.balance;
            saveDB();
            res.writeHead(200, { 'Content-Type': 'application/json' });
            res.end(JSON.stringify({ status: 'success', account: db.account }));
        });
        return;
    }

    if (url === '/api/transactions' && req.method === 'GET') {
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify(db.transactions));
        return;
    }

    if (url === '/api/admin/add_transaction' && req.method === 'POST') {
        let body = '';
        req.on('data', chunk => body += chunk);
        req.on('end', () => {
            const t = JSON.parse(body);
            const amountChange = t.is_income ? Math.abs(t.amount) : -Math.abs(t.amount);
            db.account.balance += amountChange;

            const now = new Date();
            const dateStr = t.date_str || `${String(now.getDate()).padStart(2, '0')}.${String(now.getMonth() + 1).padStart(2, '0')}.${now.getFullYear()}`;
            const timeStr = t.time_str || `${String(now.getHours()).padStart(2, '0')}:${String(now.getMinutes()).padStart(2, '0')}:${String(now.getSeconds()).padStart(2, '0')}`;
            const refNo = t.fast_ref_no || `1542${now.getTime().toString().slice(-6)}`;

            const newTx = {
                id: Date.now(),
                account_id: 1,
                transaction_type: t.transaction_type || "HESAPTAN FAST",
                title: t.sender_receiver_name,
                sender_receiver_name: t.sender_receiver_name,
                iban: t.iban || db.account.iban,
                amount: amountChange,
                balance_after: db.account.balance,
                category: t.is_income ? "Gelen Transfer" : "Para Transferi",
                date_str: dateStr,
                time_str: timeStr,
                fast_ref_no: refNo,
                commission: t.commission || 7.97,
                bsmv: t.bsmv || 0.40,
                description: t.description || `${t.sender_receiver_name}-FAST-CEP`,
                is_income: t.is_income
            };

            db.transactions.unshift(newTx);
            saveDB();

            res.writeHead(200, { 'Content-Type': 'application/json' });
            res.end(JSON.stringify({ status: 'success', new_balance: db.account.balance, transaction: newTx }));
        });
        return;
    }

    if (url.startsWith('/api/admin/transaction/') && req.method === 'DELETE') {
        const id = parseInt(url.split('/').pop());
        db.transactions = db.transactions.filter(t => t.id !== id);
        saveDB();
        res.writeHead(200, { 'Content-Type': 'application/json' });
        res.end(JSON.stringify({ status: 'deleted' }));
        return;
    }

    if (url.startsWith('/extracted_assets/')) {
        const filePath = path.join(__dirname, 'mobile_app', url);
        if (fs.existsSync(filePath)) {
            res.writeHead(200);
            fs.createReadStream(filePath).pipe(res);
            return;
        }
    }

    // --- STATIC FILE ROUTES ---
    if (url === '/' || url === '/app') {
        fs.readFile(path.join(__dirname, 'mobile_app', 'index.html'), (err, content) => {
            res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
            res.end(content);
        });
        return;
    }

    if (url === '/admin') {
        fs.readFile(path.join(__dirname, 'admin_panel', 'index.html'), (err, content) => {
            res.writeHead(200, { 'Content-Type': 'text/html; charset=utf-8' });
            res.end(content);
        });
        return;
    }

    res.writeHead(404);
    res.end('Not Found');
});

server.listen(PORT, () => {
    console.log(`====================================================`);
    console.log(`  Yapı Kredi Mobil Clone Canlı Sunucusu Çalışıyor!  `);
    console.log(`====================================================`);
    console.log(`• Yapı Kredi Mobil Arayüzü : http://127.0.0.1:8000/app`);
    console.log(`• Web Admin Yönetim Paneli : http://127.0.0.1:8000/admin`);
    console.log(`• REST API Endpoint       : http://127.0.0.1:8000/api/account`);
    console.log(`====================================================\n`);
});
